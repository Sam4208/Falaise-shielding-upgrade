56 //! Class holding a set of key-value properties
57 /*!
58  * Provides a convenient adaptor interface over datatools::properties,
59  * targeted at developers of modules for Falaise needing key-value storage
60  * and validation of input parameters. For example, say we program a module
61  * that requires two parameters, or properties, to implement its processing:
62  *
63  * * ```cpp
64  * class MyModule
65  * {
66  * public:
67  * MyModule(falaise::property_set const& config, datatools::service_manager& services){
68  * firstParam = config.get<int>("firstParam");
69  * secondParam = config.get<double>("secondParam");
70  * }
71  *
72  * falaise::processing::status process(datatools::things&) {
73  * std::cout << firstParam * secondParam << std::endl;
74  * ...
75  * }
76  *
77  * private:
78  * int firstParam = 1;
79  * double secondParam = 3.14;
80  * };
81  *
82  * Users of this module can change the `firstParam` and `secondParam` properties when
83  * using it in `flreconstruct` by configuring it in their pipeline script:
84  *
85  * ```
86  * [name="MyModule", type="MyModule"]
87  * firstParam : int = 2
88  * secondParam : real = 4.13
89  * ```
90  *
91  * The module would then print "8.26" when the `process` member function is called. In particular,
92  * it allows module authors to validate this user supplied input as needed to ensure correct
93  * running of the module. As detailed below, this may include required vs optional values, type
94  * checking, and value checking.
95  *
96  *
97  * A property_set may be created and used directly as
98  *
99  * ```cpp
100  * falaise::property_set ps{};
101  *
102  * // Store an integer at key "a"
103  * ps.put("a",2);
104  *
105  * // Get the integer
106  * auto x = ps.get<int>("a");
107  *
108  * // Replace the integer with a double
109  * ps.put_or_replace("a",3.14);
110  *
111  * // Get a string, setting a default value in case key "b" doesn't exist
112  * std::string s = ps.get<std::string>("b", "hello");
113  * std::cout << s << std::endl; // prints "hello"
114  *
115  * // Assign value a variable only if key exists
116  * std::string myDefault = "foo";
117  *
118  * // o.k....
119  * myDefault = ps.get<std::string>("nonexistent", myDefault);
120  *
121  * // better...
122  * ps.assign_if("nonexistent", myDefault);
123  * std::cout << myDefault << std::endl; // prints "foo"
124  *
125  * ps.put("myDef", "bar");
126  * ps.assign_if("myDef", myDefault);
127  * std::cout << myDefault << std::endl; // prints "bar"
128  * ```
129  *
130  * or it can be constructed from an existing instance of @ref datatools::properties
131  *
132  * ```cpp
133  * void example(datatools::properties const& x) {
134  * falaise::property_set ps{x};
135  *
136  * // ... use ps ...
137  * }
138  * ```
139  *
140  * or from a @ref datatools::properties file
141  *
142  * ```cpp
143  * void example(std::string const& file) {
144  * falaise::property_set ps{};
145  * falaise::make_property_set(file, ps);
146  *
147  * // ... use ps ...
148  * }
149  * ```
150  *
151  * or from any input stream, e.g
152  *
153  * ```cpp
154  * void example(std::istream& is) {
155  * falaise::property_set ps{};
156  * falaise::make_property_set(is, ps);
157  * // ... use ps ...
158  * }
159  * ```
160  *
161  * Storage and retrieval of parameters is typesafe in the sense that:
162  *
163  * - You can only @ref put values that are of type
164  * - int
165  * - double
166  * - bool
167  * - std::string
168  * - @ref falaise::path
169  * - @ref falaise::quantity_t
170  * - Plus any type listed in @ref falaise_units
171  * - @ref falaise::property_set
172  * - std::vector<int>
173  * - std::vector<double>
174  * - std::vector<bool>
175  * - std::vector<std::string>
176  * - You can only @ref get values of the above types, and a @ref wrong_type_error
177  * will be thrown if you try to retrieve a value with a different type, e.g
178  * ``` cpp
179  * ps.put("a", 2); // an integer by type inference (2. or 2.0 would be double)
180  * auto x = ps.get<double>("a"); // throws wrong_type_error
181  *
182  * // Or you can check manually
183  * if (ps.is_key_to<int>("a")) {
184  * std::cout << "is int\n"; // prints "is int"
185  * }
186  * ```
187  *
188  * Support for @ref path and @ref quantity_t is provided to support validation
189  * of input configuration scripts. These types model the `path` and `dimension`
190  * decorators for properties parameters:
191  *
192  * ```conf
193  * # this is a true filesystem path
194  * foo : string as path = "${HOME}/something"
195  *
196  * # this is a true physical quantity with dimension "length"
197  * bar : real as length = 3.14 m
198  * ```
199  *
200  * When extracting parameters, you can ensure that that you have a true path
201  * or physical quantity of the right dimension:
202  *
203  * ```cpp
204  * void configure_me(property_set const& ps) {
205  * // throws wrong_type_error if value at "foo" is not a path
206  * auto foo = ps.get<falaise::path>("foo");
207  *
208  * // throws wrong_type_error if value at bar is not a physical quantity with 'length' dimension
209  * auto bar = ps.get<falaise::length_t>("bar");
210  *
211  * ...
212  * }
213  * ```
214  *
215  * A full list of available quantity types is provided in the documentation for @ref quantity_t
216  *
217  * Support for @ref put and @ref get of @ref property_set values is provided to
218  * support @ref datatools::properties constructs of the form:
219  *
220  * ```conf
221  * plain : string = "A plain property"
222  * table.foo : string = "A string property in a 'table'"
223  * table.bar : int = 42
224  * ```
225  *
226  * The `foo` and `bar` properties an be extracted either by their full keys:
227  *
228  * ```cpp
229  * auto foo = ps.get<std::string>("table.foo");
230  * auto bar = ps.get<int>("table.bar);
231  * ```
232  *
233  * or by getting the `table` key as a @ref property_set, then its properties
234  * which will be stored using their subkey names:
235  *
236  * ```cpp
237  * auto table = ps.get<falaise::property_set>("table");
238  * auto foo = table.get<std::string>("foo");
239  * auto bar = table.get<int>("bar);
240  * ```
241  *
242  * This can be extended to further levels of nesting if required. However,
243  * note that @ref datatools::properties configuration of the form:
244  *
245  * ```conf
246  * table : string = "Not a valid table"
247  * table.x : int = 11
248  * table.y : int = 22
249  * ```
250  *
251  * are not considered get-able as `property_set`s. This is due to the ambiguity
252  * of the `table` key which @ref property_set regards as an atomic property
253  * of type `std::string`. Trying to extract the above `table` key as a `property_set`
254  * would result in a `wrong_type_error` exception:
255  *
256  *
257  * ```cpp
258  * // throws falaise::wrong_type_error
259  * auto table = ps.get<falaise::property_set>("table");
260  * ```
261  *
262  * The `table.x` and table.y` keys in the example can only be extracted
263  * via these fully qualified keys. It is therefore strongly recommended that
264  * you structure your configuration in the "pure table" form above for clarity
265  * and ease of use.
266  *
267  * The default value form of @ref get or the @ref assign_if member function can be used to
268  * implement optional configuration, for example
269  *
270  * ```cpp
271  * void configure_me(property_set const& ps) {
272  * // throws missing_key_error if ps doesn't have "foo"
273  * auto requiredParam = ps.get<int>("foo");
274  *
275  * // sets optionalParam to the value held at "answer", or 42 if ps doesn't have an "answer" key
276  * auto optionalParam = ps.get<int>("answer",42);
277  *
278  * // In many use cases, we may already have a parameter set to its default value
279  * // e.g. class member initialization. We could write...
280  * int defaultParameter = 42;
281  * defaultParameter = ps.get<int>("answer", defaultParameter);
282  *
283  * // ... but assign_if allows the clearer form
284  * ps.assign_if("answer", defaultParameter);
285  *
286  * ...
287  * }
288  * ```
289  *
290  * Limited validation of extracted values is available through the @ref get_if and @ref assign_if
291  * interfaces. These forms allow a value meeting the [UnaryPredicate](https://en.cppreference.com/w/cpp/named_req/Predicate)
292  * concept to be supplied that can validate the value as needed. Falaise does not supply
293  * any checks "out the box", but they are easy to write as lambdas, free functions, or
294  * function objects. For example:
295  *
296  * ```cpp
297  * // Use a lambda as simple example, but any callable taking value and returning bool works
298  * auto int_predicate = [](const int x) {
299  * // Return true if the value is between 0 and 8
300  * return (x >= 0) && (x <= 8);
301  * }
302  *
303  * falaise::property_set ps;
304  * ps.put("good", 5);
305  * ps.put("bad", -4);
306  *
307  * // Required key forms
308  * auto good = ps.get_if<int>("good", int_predicate); // fine, good = 5
309  * auto bad = ps.get_if<int>("bad", int_predicate); // throws falaise::unmet_predicate_error exception
310  *
311  * // Optional key forms
312  * auto goodDefault = ps.get_if<int>("nokey", 2, int_predicate); // fine
313  * auto badDefault = ps.get_if<int>("good", 293, int_predicate); // throws, default value must meet constraint!
314  * auto badValue = ps.get_if<int>("bad", 6, int_predicate); // throws, default is good, value is bad!
315  *
316  * // Assign_if forms
317  * int myParam = 2;
318  * ps.assign_if("nokey", myParam, int_predicate); // fine, myParam is 2
319  * ps.assign_if("good", myParam, int_predicate); //fine, myParam is now 5
320  * ps.assign_if("bad", myParam, int_predicate); //throws, myParam is still 5
321  * int myParam = 84632;
322  * ps.assign_if("anykey", myParam, int_predicate); //throws, myParam must meet constraint!
323  * ```
324  *
325  * Combined, these various ways of using property_set allow simple and reliable usage
326  * for configuration and configuration validation in Falaise modules.
327  *
328  * \sa path
329  * \sa quantity
330  * \sa datatools::properties
331  * \sa datatools::units
332  */
