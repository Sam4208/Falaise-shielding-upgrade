32  //! Get or add instance of T at supplied key in input event_record
33  /*!
34  * A simple convenience wrapper around a typical use case in Falaise
35  * modules: retrieve a non-const reference to the value of type T stored
36  * at "key" in the event_record, inserting a new value of that type at "key"
37  * if "key is not present. In code this results in the boilerplate:
38  *
39  * ```cpp
40  * // Usually in the "process" member function of a Falaise module
41  * SomeDataType* x{nullptr};
42  *
43  * if(event.has(key)) {
44  * x = event.grab<SomeDataType>(key); // throws if key does not hold a value of type T
45  * }
46  * else {
47  * x = event.add<SomeDataType>(key);
48  * }
49  *
50  * // do something with x
51  * ```
52  *
53  * This wrapper function reduces the above to:
54  *
55  * ```
56  * auto x = getOrAddToEvent<SomeDataType>(key, event);
57  * ```
58  *
59  * It additionally checks at compile time that the supplied type (`SomeDataType` here)
60  * meets the requirements to be stored in the event.
61  *
62  * \tparam T type of value to be retrieved or added
63  * \param[in] key key for value to find
64  * \param[in] event event_record to search in
65  * \returns reference to value held at key
66  * \throws datatools::bad_things_cast if value at key is not of type T
67  */
68  template <typename T>
69  T & getOrAddToEvent(std::string const & key, event_record & event)
70  {
71  static_assert(std::is_base_of<datatools::i_serializable, T>::value,
72  "snedm::event_record can only store types derived from datatools::i_serializable");
73  if (event.has(key)) {
74  return event.grab<T>(key);
75  }
76  return event.add<T>(key);
77  }

79  //! Add instance of T at supplied key in event_record unless key exists
80  /*!
81  * A simple convenience wrapper around a typical use case in Falaise modules:
82  * add a new value to the event at "key" only if the "key" is not already in use
83  *
84  * ```cpp
85  * // Usually in the "process" member function of a Falaise module
86  * auto x = addToEvent<SomeDataType>("mydata",event);
87  *
88  * // use x
89  * ```
90  *
91  * Whilst the event_record type has an `add` member function of its own, it is wrapped
92  * here to allow:
93  *
94  * - A clearer existing_product_key exception to be thrown if the key already exists
95  * - A compile time check that the type of data being stored meets the requirements to
96  * be stored in the event
97  *
98  * \tparam T type of value to be added
99  * \param[in] key key for value
100  * \param[in] event snedm::event_record to add the value to
101  * \returns reference to created value held at key
102  * \throws snedm::existing_product_key if key already exists
103  */
104  template <typename T>
105  T & addToEvent(std::string const & key, event_record & event)
106  {
107  static_assert(std::is_base_of<datatools::i_serializable, T>::value,
108  "snedm::event_record can only store types derived from datatools::i_serializable");
109  if (event.has(key)) {
110  throw existing_product_key{"snedm:event_record '" + event.get_name() +
111  "' already holds product '" + key + "'"};
112  }
113  // add will throw if the product exists, but check
114  // explicitly to raise a clearer exception
115  return event.add<T>(key);
116  }
