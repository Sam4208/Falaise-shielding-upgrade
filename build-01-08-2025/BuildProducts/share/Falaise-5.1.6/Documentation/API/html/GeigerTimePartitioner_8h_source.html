45  /// \brief A pre-clusterizer of Geiger hits for the SuperNEMO detector
46  /** This algorithm aims to group the Geiger hits in a given SuperNEMO event record
47  * using some simple clustering criteria:
48  * - prompt hits are grouped within a prompt pre-cluster if they lie on the same side of
49  * the source foil,
50  * - delayed hits are grouped within a delayed pre-cluster if they lie on the same side of
51  * the source foil and are in a 10 us time coincidence window.
52  * Only two prompt clusters can exists (maximum: one per side of the source foil).
53  * There is no limitation on the number of delayed clusters.
54  *
55  * The input of the algorithm is implemented through the template
56  * TrackerPreClustering::input_data<> class which is basically a collection of Geiger hits.
57  *
58  * The output of the algorithm is implemented through the template
59  * TrackerPreClustering::output_data<> class which stores collections of prompt and delayed
60  * clusters. By default the tracking chamber is considered as splitted in two parts.
61  *
62  */
63

122  /// A functor for handle on tracker hits that perform a comparison by delayed time
123  template <class Hit>
124  struct DelayTimeComparator
125  {
126  public:
127  /// Main comparison method(less than) : require non null handles and non-Nan delayed times
128  bool operator()(const Hit * lhs, const Hit * rhs) const {
129  if (!lhs->has_delayed_time() || !rhs->has_delayed_time()) {
130  return false;
131  }
132  return lhs->get_delayed_time() < rhs->get_delayed_time();
133  }
134  };

136  template <typename Hit>
137  GeigerHitTimePartition<Hit>
138  GeigerTimePartitioner::partition(const GeigerHitPtrCollection<Hit> & input)
139  {
140  using HitCollection = GeigerHitPtrCollection<Hit>;
141 
142  // Collection of hit per half-chamber :
143  HitCollection prompt_hits[2];
144  HitCollection delayed_hits[2];
145 
146  GeigerHitTimePartition<Hit> oput;
147 
148  // Partition input into ignored (bad data), prompt, and delayed sets
149  // Most of the complexity here comes from the optional partition by side...
150  for (const Hit * hitref : input) {
151  if (hitref == nullptr) {
152  continue;
153  }
154  if (hitref->is_sterile() || hitref->is_noisy()) {
155  oput.ignoredHits.push_back(hitref);
156  continue;
157  }
158 
159  int side = hitref->get_side();
160 
161  if (hitref->is_prompt()) {
162  if (classifiesPromptHits()) {
163  if (side == 0 || side == 1) {
164  int effective_side = isSplitChamber() ? side : 0;
165  prompt_hits[effective_side].push_back(hitref);
166  } else {
167  oput.ignoredHits.push_back(hitref);
168  }
169  } else {
170  oput.ignoredHits.push_back(hitref);
171  }
172  continue;
173  }
174 
175  if (hitref->is_delayed()) {
176  if (classifiesDelayedHits()) {
177  if (side == 0 || side == 1) {
178  int effective_side = isSplitChamber() ? side : 0;
179  delayed_hits[effective_side].push_back(hitref);
180  } else {
181  oput.ignoredHits.push_back(hitref);
182  }
183  } else {
184  oput.ignoredHits.push_back(hitref);
185  }
186  continue;
187  }
188  }
189 
190  if (classifiesPromptHits()) {
191  unsigned int max_side = isSplitChamber() ? 2 : 1;
192 
193  // For each side of the tracking chamber, we collect one unique candidate time-cluster of prompt
194  // hits.
195  for (unsigned int side = 0; side < max_side; side++) {
196  if (prompt_hits[side].size() == 1) {
197  oput.ignoredHits.push_back(prompt_hits[side][0]);
198  } else if (prompt_hits[side].size() > 1) {
199  HitCollection new_prompt_cluster(prompt_hits[side]);
200  oput.promptClusters.push_back(new_prompt_cluster);
201  }
202  }
203  }
204 
205  if (classifiesDelayedHits()) {
206  // For each side of the tracking chamber, we try to collect some candidate time-clusters of
207  // delayed hits. The aggregation criterion uses a time-interval of width
208  // 'maxDelayedHitTimeGap_' (~10usec).
209  unsigned int max_side = isSplitChamber() ? 2 : 1;
210 
211  for (unsigned int side = 0; side < max_side; side++) {
212  if (delayed_hits[side].empty()) {
213  continue;
214  }
215 
216  // Single hits are not clustered
217  if (delayed_hits[side].size() == 1) {
218  oput.ignoredHits.push_back(delayed_hits[side][0]);
219  continue;
220  }
221 
222  // Otherwise, sort in order of delay time and window cluster
223  std::sort(delayed_hits[side].begin(), delayed_hits[side].end(), DelayTimeComparator<Hit>{});
224 
225  // Pick up the first time-orderer delayed hit on this side of the source foil as
226  // the start of a forseen cluster :
227  const Hit* startHit = delayed_hits[side].at(0);
228  HitCollection* currentCluster = nullptr;
229 
230  // Traverse remaining delayed hits from this side :
231  for (size_t i = 1; i < delayed_hits[side].size(); i++) {
232  const Hit* currentHit = delayed_hits[side].at(i);
233  // Time of current hit must be within the window of the current
234  // cluster's start time
235  bool isOutsideWindow = (currentHit->get_delayed_time() >
236  (startHit->get_delayed_time() + maxDelayedHitTimeGap_));
237 
238  if (isOutsideWindow) {
239  // With no existing cluster, the earlier hit is isolated
240  if (currentCluster == nullptr) {
241  oput.ignoredHits.push_back(startHit);
242  }
243  // Make the current hit the new start point and require a new cluster:
244  startHit = currentHit;
245  currentHit = nullptr;
246  currentCluster = nullptr;
247  } else {
248  // cluster the two hits, creating a new cluster if needed
249  if (currentCluster == nullptr) {
250  // New time cluster
251  oput.delayedClusters.push_back(HitCollection{});
252  currentCluster = &(oput.delayedClusters.back());
253  currentCluster->push_back(startHit);
254  }
255  // Record the current hit in the current time-cluster :
256  currentCluster->push_back(currentHit);
257  }
258  }
259  }
260  }
261 
262  return oput;
263  }
