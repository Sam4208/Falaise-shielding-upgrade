35  /// Supported categories of vertex
36  enum vertex_category_type {
37  VERTEX_CATEGORY_UNDEF = 0x0, ///< Undefined category
38  VERTEX_CATEGORY_ON_SOURCE_FOIL = datatools::bit_mask::bit00, ///< Maybe a source strip or pad bulk physical volume
39  VERTEX_CATEGORY_ON_MAIN_CALORIMETER = datatools::bit_mask::bit01, ///< The entrance window of a main calorimeter block (front)
40  VERTEX_CATEGORY_ON_X_CALORIMETER = datatools::bit_mask::bit02, ///< The entrance window of a main X-calorimeter block
41  VERTEX_CATEGORY_ON_GAMMA_VETO = datatools::bit_mask::bit03, ///< The bulk volume of a gamma veto block (no way to locate the entrance window)
42  VERTEX_CATEGORY_ON_WIRE = datatools::bit_mask::bit04, ///< The vicinity of a wire in the tracking chamber
43  VERTEX_CATEGORY_ON_CALIBRATION_SOURCE = datatools::bit_mask::bit05, ///< A calibration source (Bi207)
44  VERTEX_CATEGORY_ON_SOURCE_GAP = datatools::bit_mask::bit06, ///< A gap between source pads (vertex reconstructed in a region of the source plane without material !)
45  VERTEX_CATEGORY_ON_REFERENCE_SOURCE_PLANE = datatools::bit_mask::bit07, ///< Fake vertex reconstructed on the reference source plane
46  VERTEX_CATEGORY_IN_GAS = datatools::bit_mask::bit08 ///< Vertex reconstructed in the tracker gas deep volume
47  };
