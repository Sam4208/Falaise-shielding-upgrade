32 //! Class representing a filesystem path as held by a falaise::property_set
33 /*!
34  * Filesystem paths can be defined in datatools::properties syntax as:
35  *
36  * ```ini
37  * mypath : string as path = "/some/path"
38  * ```
39  *
40  * Parsing of the properties file will expand any mount points or environment
41  * variables so that fetching the path will return a `std::string` containing
42  * the resolved absolute path.
43  *
44  * @ref path provides a simple type to distinguish raw `std::string` from
45  * explicit paths allowing users to validate that a falaise::property_set value
46  * is a true path. It is nothing more than a simple holder of the std::string for
47  * the absolute path and may be used as:
48  *
49  * ```cpp
50  * void configure_me(property_set const& ps) {
51  * auto p = ps.get<falaise::path>("mypath"); // throws if "mypath" is not a path value
52  * std::cout << p << std::endl; // prints absolute path
53  * }
54  * ```
55  *
56  * A @ref path can be constructed from any string whose value is a valid path
57  * as understood by @ref datatools::fetch_path_with_env. Examples are:
58  *
59  * - An absolute path, e.g. `auto x = falaise::path{"/foo/bar/baz.txt"};`
60  * - A relative path, e.g. `auto x = falaise::path{"bar/baz.txt"};`
61  *
62  * The string may also include environment variables, e.g. `$HOME` or `${HOME}`,
63  * which will be expanded if they exist, otherwise a @ref invalid_path_error
64  * is thrown. When running in a program that provides the @ref datatools::kernel
65  * service, paths can also be supplied as:
66  *
67  * - A resource path, e.g. `@falaise:foo/bar.txt`
68  * - A @ref datatools::urn, e.g. `urn:subject:object`
69  *
70  * Note that in all cases, @ref path *does not* check that a file exists at the location
71  * specified by the path. If you require more specific filesystem functionality, you
72  * should use the value of path in a full library such as
73  * [boost::filesystem](https://www.boost.org/doc/libs/1_69_0/libs/filesystem/doc/index.htm) or
74  * C++17's [std::filesystem](https://en.cppreference.com/w/cpp/filesystem) if available.
75  *
76  * \sa property_set
77  * \sa datatools::fetch_path_with_env
78  */
79 class path {

81  //! Default constructor

85  /*!
86  * Not declared explicit as we allow conversions for compatibility with
87  * existing use of filesystem paths as std::string.
88  *

90  * \throws invalid_path_error if p is not a valid path string
91  */
92  path(std::string const& p);

102

118  */
119  bool operator!=(std::string const& other) const { return !(*this == other); }
120 
121  private:

123 };
