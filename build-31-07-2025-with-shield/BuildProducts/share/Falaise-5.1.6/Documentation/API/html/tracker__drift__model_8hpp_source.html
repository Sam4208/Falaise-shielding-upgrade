42  /// \brief Cell quarter category
43  ///
44  /// A tracker cell is divided in 4 quarters (see DocDb #5839: p. 24, fig. 15)
45  /// corresponding to regions with
46  /// specific electric field conditions, mainly depending on the environnement
47  /// of the cell: neighbouring cells with HV applied or not, edge of the tracking chamber...
48  /// Each quarter is identified in a given cell with a conventional index from 0 to 3, with respect to
49  /// the cell reference frame in the horizontal XY plane:
50  ///
51  /// ^ Y+
52  /// :
53  /// : a cell and its 4 quarters
54  /// +---------------+
55  /// | . |
56  /// | 1 . 0 |
57  /// | . |
58  /// ----| - - - o - - - |-----> X+
59  /// | . |
60  /// | 2 . 3 |
61  /// | . |
62  /// +---------------+
63  /// :
64  ///
65  /// Four distinct categories of cell quarter are defined (see DocDB #5839):
66  ///
67  /// - "centre" (ce) : a cell quarter with HV applied on both first neighbour cells
68  /// Example:
69  ///
70  /// +-------+-------+--
71  /// |HV any | HV on |
72  /// | o | o |
73  /// | | |
74  /// +-------+-------+--
75  /// | HV on |ce : |
76  /// | o |---o---| <--- a cell and 3 of its neighbour cells
77  /// | | : |
78  /// +-------+-------+--
79  /// | | |
80  ///
81  /// - "edge" (ed) : a cell quarter with HV applied on only one neighbour cell, and not on the other one
82  ///
83  /// Example:
84  /// +-------+-------+
85  /// |HV any | HV on |
86  /// | o | o |
87  /// | | |
88  /// +-------+-------+
89  /// | HV off|ed : |
90  /// | o |---o---|
91  /// | | : |
92  /// +-------+-------+
93  ///
94  /// - "corner" (co) : a cell quarter with HV not applied on both first neighbour cells
95  ///
96  /// Example:
97  /// +-------+-------+
98  /// |HV any | HV off|
99  /// | o | o |
100  /// | | |
101  /// +-------+-------+
102  /// | HV off|co : |
103  /// | o |---o---|
104  /// | | : |
105  /// +-------+-------+
106  ///
107  enum class cell_quarter_category {
108  none, ///< Undefined category
109  centre, ///< Centre quarter (both first neighbour cells with HV on)
110  edge, ///< Edge quarter (one neighbour cell with HV on, and one with HV off or missing)
111  corner ///< Corner quarter (both first neighbour cells with HV off or missing)
112  };

116  /// \brief Cell quarter edge category
117  ///
118  /// This information encodes the direction of the neighbour cell with HV OFF in case of the 'edge' category.
119  /// This direction is relative to the official reference frame in the horizontal (XY) plane. This information
120  /// is important to properly take into account the asymmetry of the electric field for an 'edge' quarter.
121  ///
122  /// Examples:
123  ///
124  /// * Edge category with "back" direction:
125  ///
126  /// ^ Y+
127  /// :
128  /// right
129  /// +-------+-------+
130  /// | | HV on |
131  /// | o | o |
132  /// | | |
133  /// back +-------+-------+ front -> X+
134  /// |HV off <ed : |
135  /// | o |---o---|
136  /// | | : |
137  /// +-------+-------+
138  /// left
139  ///
140  /// * Edge category with "right" direction:
141  ///
142  /// ^ Y+
143  /// :
144  /// right
145  /// +-------+-------+
146  /// | | HV off|
147  /// | o | o |
148  /// | | |
149  /// back +-------+-^-----+ front -> X+
150  /// |HV on |ed : |
151  /// | o |---o---|
152  /// | | : |
153  /// +-------+-------+
154  /// left
155  ///

159  front,

166  /// This bit mask locate the positions of neighbourg cells with HV off with respect
167  /// to a given cell used as a reference:
168  ///
169  /// Example: for a cell with two of its four neighbour cells with HV off, the associated
170  /// 'neighbour_cells_off_pattern' bitset is thus: ``off_cell_direction::front | off_cell_direction::right``
171  ///
172  /// ^ Y+
173  /// :
174  /// right
175  /// +-------+-------+-------+
176  /// | | HV off| |
177  /// | o | o | o |
178  /// | | | |
179  /// +-------+-------+-------+
180  /// | HV on | : | HV off|
181  /// back | o |---o---| o | front -> X+
182  /// | | : | |
183  /// +-------+-------+-------+
184  /// | | HV on | |
185  /// | o | o | . |
186  /// | | | o |
187  /// +-------+-------+-------+
188  /// left
189  ///
190  struct off_cell_direction {
191  static const std::uint32_t back = datatools::bit_mask::bit00; ///< 0001 == 000B
192  static const std::uint32_t front = datatools::bit_mask::bit01; ///< 0010 == 00F0
193  static const std::uint32_t left = datatools::bit_mask::bit02; ///< 0100 == 0L00
194  static const std::uint32_t right = datatools::bit_mask::bit03; ///< 1000 == R000
195  };

197  /// A cell OFF-pattern consists in a bitset of 4 bits using the conventional *RLFB* order.
198  ///
199  /// Each bit corresponds to the E-field OFF status of some neighbour cells
200  /// in a given direction with respect to a given cell.
201  ///
202  /// right
203  /// ^ (Y+)
204  /// :
205  /// back --[cell]--> front (X+)
206  /// :
207  /// left
208  ///
209  /// A "off_cell_direction" bit is set when the E-field is OFF or also if
210  /// there is no cell in this direction (edge or corner of a tracker half-chamber)
211  ///
212  /// Examples:
213  ///
214  /// rlfb
215  /// 0010 : front cell's HV is OFF or (side=0,layer=0) or (side=1,layer=8)
216  /// 0110 : front and left cells' HVs are OFF or (side=0,layer=0,row=0) or (side=1,layer=8,row=0)
217  ///
218  typedef std::uint32_t neighbour_cells_off_pattern;

345  /// \brief Return the cell info object associated to a given tracker cell and timepoint.
346  ///
347  /// The *cell info* object depends on the placement of the cell in the tracker geometry
348  /// (central cell, cell near an edge, cell at a corner) but also, after the "betsy" drift model,
349  /// on the HV status of its first neighbourg cells (back, front, left, right).
350  /// The geometry layout of the cell is of course not mutable. However, the HV status (on/off) of neighbourg cells
351  /// may change during the lifetime of the experiment. As a consequence, the shape of the electric field
352  /// in some cell quarters can be modified. A request to the *tracker cell status service* can be done
